IterativeTraversal ofBinaryTree
class Node:
def __init__(self, data, left=None, right=None):
self.data, self.left, self.right = data, left, right
def inorder(r):
st, res = [], []
while r or st:
while r:
st.append(r); r = r.left
r = st.pop(); res.append(r.data); r = r.right
return res
def preorder(r):
st, res = [r], []
while st:
n = st.pop(); res.append(n.data)
if n.right: st.append(n.right)
if n.left: st.append(n.left)
return res
def postorder(r):
st, res = [r], []
while st:
n = st.pop(); res.append(n.data)
if n.left: st.append(n.left)
if n.right: st.append(n.right)
return res[::-1]
root = Node("A", Node("B", Node("D"), Node("E")), Node("C", None, Node("F")))
print("Binary Tree:")
print(" A")
print(" / \\")
print(" B C")
print(" / \\ \\")
print(" D E F")
print("\nInorder:", inorder(root))
print("Preorder:", preorder(root))
print("Postorder:", postorder(root))

Count Nodes (Total, Leaf,Internal)
class Node:
def __init__(self, data, left=None, right=None):
self.data, self.left, self.right = data, left, right
def count(r):
if not r:
return 0, 0, 0
if not r.left and not r.right:
return 1, 1, 0
t1, l1, i1 = count(r.left)
t2, l2, i2 = count(r.right)
return 1 + t1 + t2, l1 + l2, 1 + i1 + i2
root = Node("A", Node("B", Node("D"), Node("E")), Node("C", None, Node("F")))
print("Binary Tree:")
print(" A")
print(" / \\")
print(" B C")
print(" / \\ \\")
print(" D E F")
total, leaf, internal = count(root)
print("\nTotal Nodes:", total)
print("Leaf Nodes:", leaf)
print("Internal Nodes:", internal)

Minimum and Maximum inBinaryTree
class Node:
def __init__(self, data, left=None, right=None):
self.data, self.left, self.right = data, left, right
def values(r):
return [r.data] + values(r.left) + values(r.right) if r else []
root = Node(50, Node(30, Node(20), Node(40)), Node(70, None, Node(80)))
print("Binary Tree:")
print(" 50")
print(" / \\")
print(" 30 70")
print(" / \\ \\")
print(" 20 40 80")
v = values(root)
print("\nMinimum:", min(v))
print("Maximum:", max(v))

Binary SearchTree:Insert and Search
class Node:
def __init__(self, data, left=None, right=None):
self.data, self.left, self.right = data, left, right
def insert(r, x):
if not r:
return Node(x)
if x < r.data:
r.left = insert(r.left, x)
else:
r.right = insert(r.right, x)
return r
def search(r, key):
if not r:
return False
if key == r.data:
return True
return search(r.left if key < r.data else r.right, key)
root = None
for x in [50, 30, 70, 20, 40, 60, 80]:
root = insert(root, x)
print("Binary Search Tree:")
print(" 50")

print("Element found" if search(root, key) else "Element not found")
