Set Cover
universe = {1, 2, 3, 4, 5}
sets = {"S1": {1, 2, 3}, "S2": {2, 4}, "S3": {3, 4, 5}, "S4": {5}}
left, chosen = set(universe), []
while left:
best = max(sets, key=lambda s: len(sets[s] & left))
chosen.append(best)
left -= sets[best]
print("Selected sets:", chosen)
print("All elements are covered.")

Subset Sum
numbers = sorted([3, 5, 7, 10, 12], reverse=True)
target, subset, total = 15, [], 0
for x in numbers:
if total + x <= target:
subset.append(x)
total += x
print("Numbers:", numbers)
print("Target:", target)
print("Selected subset:", subset)
print("Subset sum:", total)

Vertex Cover
edges = [("A", "B"), ("A", "C"), ("B", "C"), ("B", "D"), ("C", "D")]
cover = []
while edges:
u, v = edges[0]
cover += [u, v]
edges = [e for e in edges if u not in e and v not in e]
print("Vertex Cover:", cover)
print("Number of vertices:", len(cover))
