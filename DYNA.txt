Rod Cutting
def rod_cutting(price, n):
dp = [0] * (n + 1)
for i in range(1, n + 1):
dp[i] = max(price[j - 1] + dp[i - j] for j in range(1, i + 1))
return dp[n]
print("Maximum Profit:", rod_cutting([1, 5, 8, 9], 4))

Matrix Chain Multiplication
def mcm(p):
n = len(p) - 1
dp = [[0] * (n + 1) for _ in range(n + 1)]
for L in range(2, n + 1):
for i in range(1, n - L + 2):
j = i + L - 1
dp[i][j] = min(dp[i][k] + dp[k + 1][j] + p[i - 1] * p[k] * p[j]
for k in range(i, j))
return dp[1][n]
print("Minimum Multiplication Operations:", mcm([10, 30, 5, 60]))

Longest Common Subsequence (LCS)
def lcs(a, b):
dp = [[""] * (len(b) + 1) for _ in range(len(a) + 1)]
for i in range(1, len(a) + 1):
for j in range(1, len(b) + 1):
if a[i - 1] == b[j - 1]:
dp[i][j] = dp[i - 1][j - 1] + a[i - 1]
else:
dp[i][j] = max(dp[i - 1][j], dp[i][j - 1], key=len)
return dp[-1][-1]
s = lcs("BATMAN", "CATWOMAN")
print("LCS Length:", len(s))
print("Subsequence:", s)
