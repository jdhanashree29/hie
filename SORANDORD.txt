ch = int(input("Enter your choice (1 or 2): "))

Radix Sort
n = int(input("Enter the number of elements: "))
print("Enter the elements:")
a = list(map(int, input().split()))
print("Original Array:", a)
e = 1
while max(a) // e > 0:
b = [[] for _ in range(10)]
for x in a:
b[x // e % 10].append(x)
a = [x for l in b for x in l]
e *= 10
print("Sorted Array:", a)

Bubble Sort
n = int(input("Enter the number of elements: "))
print("Enter the elements:")
a = list(map(int, input().split()))
print("Original Array:", a)
for i in range(n - 1):
for j in range(n - i - 1):
if a[j] > a[j + 1]:
a[j], a[j + 1] = a[j + 1], a[j]
print("Sorted Array:", a)

Bucket Sort
n = int(input("Enter the number of elements: "))
print("Enter the elements:")
a = list(map(int, input().split()))
print("Original Array:", a)
b = [0] * (max(a) + 1)
for x in a:
b[x] += 1
s = []
for i in range(len(b)):
s += [i] * b[i]
print("Sorted Array:", s)
