BFS
graph = {'A': ['B', 'C'], 'B': ['D', 'E'], 'C': ['F'], 'D': [], 'E': [], 'F':
[]}
def bfs(start):
visited, queue = [start], [start]
while queue:
node = queue.pop(0)
print(node, end=" ")
for n in graph[node]:
if n not in visited:
visited.append(n)
queue.append(n)
print("BFS Traversal:")
bfs('A')

DFS
graph = {'A': ['B', 'C'], 'B': ['D', 'E'], 'C': ['F'], 'D': [], 'E': [], 'F':
[]}
def dfs(node, visited):
if node not in visited:
print(node, end=" ")
visited.add(node)
for n in graph[node]:
dfs(n, visited)
print("DFS Traversal:")
dfs('A', set())

Kruskal's Algorithm
edges = [(0, 1, 10), (0, 2, 6), (0, 3, 5), (1, 3, 15), (2, 3, 4)]
edges.sort(key=lambda e: e[2])
parent = [0, 1, 2, 3]
def find(x):
while parent[x] != x:
x = parent[x]
return x
result, total = [], 0
for u, v, w in edges:
a, b = find(u), find(v)
if a != b:
result.append((u, v, w))
total += w
parent[a] = b
print("Edges in Minimum Spanning Tree:")
for u, v, w in result:
print(u, "--", v, "=", w)
print("Total Cost:", total)

Prim's Algorithm
g = [[0, 10, 6, 5], [10, 0, 0, 15], [6, 0, 0, 4], [5, 15, 4, 0]]
n = 4
visited, total = [0], 0
print("Edges in Minimum Spanning Tree:")
while len(visited) < n:
w, x, y = min((g[i][j], i, j) for i in visited
for j in range(n) if j not in visited and g[i][j])
print(x, "--", y, "=", w)
total += w
visited.append(y)
print("Total Cost:", total)

Dijkstra's Algorithm
g = [[0, 10, 6, 5], [10, 0, 0, 15], [6, 0, 0, 4], [5, 15, 4, 0]]
n = 4
dist = [999] * n
dist[0] = 0
done = [False] * n
for _ in range(n):
u = min((i for i in range(n) if not done[i]), key=lambda i: dist[i])
done[u] = True
for v in range(n):
if g[u][v] and dist[u] + g[u][v] < dist[v]:
dist[v] = dist[u] + g[u][v]
print("Shortest distances from vertex 0:")
for i in range(n):
print("0 ->", i, "=", dist[i])
