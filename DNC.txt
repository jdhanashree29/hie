2x2 Matrix Multiplication
import time
print("Enter elements of first 2x2 matrix:")
A = [list(map(int, input().split())) for _ in range(2)]
print("Enter elements of second 2x2 matrix:")
B = [list(map(int, input().split())) for _ in range(2)]
start = time.perf_counter()
for _ in range(1000000):
C = [[0] * 2 for _ in range(2)]
for i in range(2):
for j in range(2):
for k in range(2):
C[i][j] += A[i][k] * B[k][j]
end = time.perf_counter()
print("\nResult of Matrix Multiplication:")
for row in C:
print(row)
print("\nExecution Time: {:.10f} seconds".format(end - start))

Strassen Matrix Multiplication
import time
print("Enter first 2x2 matrix:")
A = [list(map(int, input().split())) for _ in range(2)]
print("Enter second 2x2 matrix:")
B = [list(map(int, input().split())) for _ in range(2)]
(a, b), (c, d) = A
(e, f), (g, h) = B
start = time.perf_counter()
for _ in range(1000000):
M1 = (a + d) * (e + h)
M2 = (c + d) * e
M3 = a * (f - h)
M4 = d * (g - e)
M5 = (a + b) * h
M6 = (c - a) * (e + f)
M7 = (b - d) * (g + h)
end = time.perf_counter()
print("\nResult Matrix:")
print([M1 + M4 - M5 + M7, M3 + M5])
print([M2 + M4, M1 - M2 + M3 + M6])
print("\nExecution Time: {:.10f} seconds".format(end - start))
