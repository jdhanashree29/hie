Hiring Problem
import random
n = 10
c = list(range(1, n + 1))
random.shuffle(c)
best, hires = n + 1, 0
print("Candidate Order:", c)
for x in c:
print("Interview Candidate Rank:", x)
if x < best:
best, hires = x, hires + 1
print(" -> Hired")
else:
print(" -> Rejected")
print("\nTotal Interviews =", n)
print("Total Hires =", hires)
